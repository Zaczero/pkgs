"""Correctness-only PyPI candidate probes; raw driver timings are not selection inputs."""

import argparse
import importlib.metadata
import json
import os
import resource
import subprocess
import sys
import time
from pathlib import Path

resource.setrlimit(resource.RLIMIT_AS, (4 * 1024**3, 4 * 1024**3))
ROOT = Path('/home/user/Source/pkgs.worktrees/benchmark-server/h2corn')
sys.path.insert(0, str(ROOT))
os.chdir(ROOT)

from bench._core import (
    BenchmarkError, allocated_listener, check_response, deadline_bound,
    ensure_large_upload_payload, ensure_static_file_payload, request_response,
    run_load, running_server, write_json,
)
from bench.bench import (
    benchmark_scenarios, configure_rtt50_profile, rtt50_scenarios,
    server_command, write_hypercorn_config,
)

parser = argparse.ArgumentParser()
parser.add_argument('--server', choices=['uvicorn', 'gunicorn'], required=True)
parser.add_argument('--profile', choices=['loopback', 'rtt50'], default='loopback')
parser.add_argument('--output', type=Path, required=True)
parser.add_argument('--multiplex-only', action='store_true')
args = parser.parse_args()
args.output.mkdir(parents=True, exist_ok=True)
executable = Path(sys.executable).parent / args.server
help_result = subprocess.run([str(executable), '--help'], capture_output=True, text=True, timeout=15)
help_text = help_result.stdout
(args.output / 'help.txt').write_text(help_text + help_result.stderr)
versions = {}
for package in ('uvicorn', 'zttp', 'gunicorn', 'starlette', 'h2'):
    try:
        versions[package] = importlib.metadata.version(package)
    except importlib.metadata.PackageNotFoundError:
        pass
supports_h2 = (
    '--http2' in help_text and 'zttp' in help_text
    if args.server == 'uvicorn'
    else '--http2-cleartext' in help_text
)
record = {
    'versions': versions, 'python': sys.version, 'executable': str(executable),
    'profile': args.profile, 'supports_prior_knowledge_h2': supports_h2,
    'selection_inputs': 'response contracts, exact completion/body counts and errors only',
    'cases': [], 'memory_limit_per_process_bytes': 4 * 1024**3,
}
if args.profile == 'rtt50':
    record['network'] = configure_rtt50_profile()
ensure_static_file_payload()
ensure_large_upload_payload()
config = args.output / 'hypercorn.toml'
write_hypercorn_config(config)
scenarios = benchmark_scenarios() if args.profile == 'loopback' else rtt50_scenarios()
if args.multiplex_only:
    scenarios = tuple(case for case in scenarios if 'multiplexed' in case.name)
try:
    for scenario in scenarios:
        row = {'scenario': scenario.name, 'type': scenario.type, 'workers': scenario.workers}
        record['cases'].append(row)
        if scenario.contract.protocol == '2' and not supports_h2:
            row.update(status='unsupported', reason='this release has no prior-knowledge cleartext HTTP/2 mode')
            continue
        case_directory = args.output / scenario.slug()
        case_directory.mkdir()
        try:
            with deadline_bound(time.monotonic() + 120):
                with allocated_listener(scenario) as (live, listener):
                    command = server_command(
                        args.server, live, hypercorn_config=config, listener_fd=listener.fileno(),
                    )
                    if args.server == 'gunicorn':
                        for option, arguments in (
                            ('--no-control-socket', 0), ('--http-parser', 1), ('--asgi-loop', 1),
                        ):
                            if option not in help_text and option in command:
                                index = command.index(option)
                                del command[index:index + arguments + 1]
                    row['command'] = command
                    with running_server(
                        command, workers=live.workers, scenario=live,
                        socket_path=live.socket_path, listener_fd=listener.fileno(),
                    ):
                        row['phase'] = 'response-before'
                        check_response(live)
                        row['worker_sample'] = json.loads(
                            request_response(live, '/__bench/identity', timeout=5),
                        )
                        if row['worker_sample']['event_loop'] != 'asyncio.unix_events':
                            raise BenchmarkError('candidate did not select standard asyncio')
                        row['phase'] = 'fixed-request-correctness'
                        if scenario.type == 'ws':
                            for _ in range(5):
                                check_response(live)
                            row.update(status='passed', phase='complete', completed=5)
                            print('PASS', scenario.name, flush=True)
                            continue
                        concurrency = scenario.concurrency or 100
                        requests = (
                            20000 if 'multiplexed' in scenario.name and args.profile == 'loopback'
                            else 2 * concurrency * scenario.http2_parallelism
                        )
                        metrics = run_load(
                            live, duration='1s', concurrency=concurrency,
                            requests=requests, evidence_path=case_directory / 'driver.json',
                        )
                        if metrics['completed'] != requests and scenario.type != 'ws':
                            raise BenchmarkError('not every fixed-count request completed')
                        row['completed'] = metrics['completed']
                        row['phase'] = 'response-after'
                        check_response(live)
            row.update(status='passed', phase='complete')
            print('PASS', scenario.name, flush=True)
        except (OSError, RuntimeError, ValueError) as error:
            row.update(status='failed', error=repr(error))
            print('FAIL', scenario.name, repr(error), flush=True)
            break
finally:
    write_json(args.output / 'selection.json', record)
